package ipnlocal

import (
	"bytes"
	"encoding/json"
	"fmt"
	"net/netip"
	"os/exec"
	"path/filepath"
	"slices"
	"strings"
	"testing"

	"tailscale.com/appc"
)

// Test-only overlay: exercise the actual Python seeder against Go-produced
// checksummed state, then restore/adopt/reset using the backend fixture.
// This is not a daemon/OS acceptance test.
func TestR29PythonSeed810(t *testing.T) {
	h := newTakeoverHarness(t, 0)
	raw, err := h.st.ReadState(connectorKey(h.key))
	mustTakeover(t, err)
	input, err := json.Marshal(map[string][]byte{string(connectorKey(h.key)): raw})
	mustTakeover(t, err)
	script, err := filepath.Abs("../../lab/appc-e2e/cli810.py")
	mustTakeover(t, err)
	code := `import ast,base64,hashlib,json,sys
source = sys.argv[1]
tree = ast.parse(open(source).read())
module = ast.Module(body=[n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='seed_control_history'],type_ignores=[])
exec(compile(module,source,'exec'))
store=json.load(sys.stdin)
prefixes=[f'192.0.{i//250+1}.{i%250+1}/32' for i in range(810)]
seed_control_history(store,'p',prefixes)
json.dump(store,sys.stdout)
`
	cmd := exec.Command("python3", "-c", code, script)
	cmd.Stdin = bytes.NewReader(input)
	var stderr bytes.Buffer
	cmd.Stderr = &stderr
	out, err := cmd.Output()
	if err != nil {
		t.Fatalf("Python seed: %v: %s", err, &stderr)
	}
	var store map[string][]byte
	mustTakeover(t, json.Unmarshal(out, &store))
	mustTakeover(t, h.st.WriteState(connectorKey(h.key), store[string(connectorKey(h.key))]))
	mustTakeover(t, h.st.WriteState(h.key+"||_routeInfo", store[string(h.key)+"||_routeInfo"]))
	retained, err := h.adapter.Current()
	mustTakeover(t, err)
	current := slices.Clone(retained)
	v, err := readConnectorIntent(h.st, h.key)
	mustTakeover(t, err)
	m := &appc.ControlProofManifest{SchemaVersion: 1, ProfileID: "p", LegacySourceID: v.Unclaimed.SourceID, Source: "r29-synthetic"}
	for i := 0; i < 810; i++ {
		ip := netip.MustParseAddr(fmt.Sprintf("192.0.%d.%d", i/250+1, i%250+1))
		current = append(current, netip.PrefixFrom(ip, 32))
		ref := appc.LegacyDNSReference{AssignedRoot: "*.example.test", Domain: fmt.Sprintf("d%d.example.test", i), IP: ip}
		m.Entries = append(m.Entries, appc.ControlProofEntry{Reference: ref, Reviewed: true, ReviewedBy: "fixture", Evidence: "synthetic", EvidenceSHA256: strings.Repeat("a", 64)})
		h.claims = append(h.claims, ref)
	}
	raw, err = json.Marshal(sortedRoutes(current))
	mustTakeover(t, err)
	mustTakeover(t, h.st.WriteState(h.key, raw))
	h.install(&v)
	// No proof means no eligibility; seeding did not create leases or grant auto ownership.
	if n := len(h.plan(true).Entries); n != 0 {
		t.Fatalf("unproved entries: %d", n)
	}
	h.manifest = m
	plan, err := importPlan(t, h, m)
	mustTakeover(t, err)
	if len(plan.Entries) != 810 {
		t.Fatalf("entries=%d", len(plan.Entries))
	}
	for _, e := range plan.Entries {
		if !e.Eligible {
			t.Fatalf("ineligible: %+v", e)
		}
	}
	mustTakeover(t, h.ac.AdoptTTL(plan.ConnectorHash, "r29-synthetic", h.claims))
	h.manifest = nil
	h.claims = nil
	reset := h.plan(false)
	if len(reset.Withdraw) != 810 {
		t.Fatalf("withdraw=%d", len(reset.Withdraw))
	}
	mustTakeover(t, h.ac.ResetTTL(reset.ConnectorHash, reset.Fence.Generation, "r29-reset"))
	actual, err := h.adapter.Current()
	mustTakeover(t, err)
	if !slices.Equal(sortedRoutes(actual), sortedRoutes(retained)) {
		t.Fatal("retained routes changed")
	}
}
